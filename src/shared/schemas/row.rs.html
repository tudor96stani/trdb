8/// Container storing the actual value stored in a row cell
9#[derive(Debug)]
10pub enum RowCellValue {
11    /// Integer value
12    Int(i32),
13    /// String value
14    String(String),
15    /// NULL value
16    NULL,
17}
18
19impl From<i32> for RowCellValue {
20    fn from(value: i32) -> Self {
21        Self::Int(value)
22    }
23}
24
25impl From<String> for RowCellValue {
26    fn from(value: String) -> Self {
27        Self::String(value)
28    }
29}
30
31impl From<&str> for RowCellValue {
32    fn from(value: &str) -> Self {
33        Self::String(value.into())
34    }
35}

37/// A single cell from a `Row`, containing both value and schema metadata
38#[derive(Debug)]
39pub struct RowCell {
40    schema_entry_id: SchemaEntryId,
41    value: RowCellValue,
42}
43
44impl RowCell {
45    /// Constructor
46    fn new(schema_entry_id: SchemaEntryId, value: RowCellValue) -> Self {
47        Self {
48            schema_entry_id,
49            value,
50        }
51    }
52
53    /// Getter for schema entry id of cell
54    pub fn schema_entry_id(&self) -> SchemaEntryId {
55        self.schema_entry_id
56    }
57
58    /// Getter for value of cell
59    pub fn value(&self) -> &RowCellValue {
60        &self.value
61    }
62}

64/// An in memory representation of a row
65#[derive(Debug)]
66pub struct Row {
67    values: HashMap<SchemaEntryId, RowCell>,
68}
69
70impl Row {
71    /// Constructor
72    fn new(values: HashMap<SchemaEntryId, RowCell>) -> Self {
73        Self { values }
74    }
75
76    /// Creates a `RowBuilder` to be used to generate a new `Row` object
77    pub fn builder(schema: &Schema) -> RowBuilder<'_> {
78        RowBuilder::new(schema)
79    }
80
81    /// Iterator over the cells
82    pub fn values(&self) -> impl Iterator<Item = &RowCell> {
83        self.values.values()
84    }
85
86    /// Returns the cell associated with the given schema entry ID, if present.
87    pub fn get(&self, schema_entry_id: &SchemaEntryId) -> Option<&RowCell> {
88        self.values.get(schema_entry_id)
89    }
90}

92/// A builder of a `Row`
93#[derive(Debug)]
94pub struct RowBuilder<'a> {
95    schema: &'a Schema,
96    values: HashMap<SchemaEntryId, RowCell>,
97}
98
99impl<'a> RowBuilder<'a> {
100    /// Creates a new `RowBuilder` - not to be used externally. Invoke the `builder()` method on `Row`
101    fn new(schema: &'a Schema) -> Self {
102        Self {
103            schema,
104            values: HashMap::new(),
105        }
106    }
107
108    /// Append a new cell to the row builder using an existing schema entry.
109    pub fn cell(mut self, schema_entry: &'a SchemaEntry, value: impl Into<RowCellValue>) -> Self {
110        let schema_entry_id = schema_entry.id();
111        let value = value.into();
112
113        assert!(
114            self.schema.columns().contains_key(&schema_entry_id),
115            "schema entry does not belong to this row schema",
116        );
117        assert!(
118            Self::value_matches_schema_entry(schema_entry, &value),
119            "row cell value does not match schema entry data type",
120        );
121
122        self.values
123            .insert(schema_entry_id, RowCell::new(schema_entry_id, value));
124        self
125    }
126
127    /// Append a new value to the row builder by column id.
128    pub fn value(self, column_id: ObjectId, value: impl Into<RowCellValue>) -> Self {
129        let schema_entry = self.schema_entry_by_column_id(column_id);
130        self.cell(schema_entry, value)
131    }
132
133    /// Append a new value to the row builder by column position.
134    pub fn value_at_position(self, position: usize, value: impl Into<RowCellValue>) -> Self {
135        let schema_entry = self.schema_entry_by_position(position);
136        self.cell(schema_entry, value)
137    }
138
139    /// Append a new value to the row builder by column name.
140    pub fn value_named(self, name: &str, value: impl Into<RowCellValue>) -> Self {
141        let schema_entry = self.schema_entry_by_name(name);
142        self.cell(schema_entry, value)
143    }
144
145    /// Builds the final immutable `Row`.
146    pub fn build(self) -> Row {
147        Row::new(self.values)
148    }
149
150    fn schema_entry_by_column_id(&self, column_id: ObjectId) -> &'a SchemaEntry {
151        self.schema
152            .columns()
153            .values()
154            .find(|entry| entry.id().column_id() == column_id)
155            .unwrap_or_else(|| panic!("schema does not contain column id {column_id}"))
156    }
157
158    fn schema_entry_by_position(&self, position: usize) -> &'a SchemaEntry {
159        self.schema
160            .columns()
161            .values()
162            .find(|entry| entry.position() == position)
163            .unwrap_or_else(|| panic!("schema does not contain column at position {position}"))
164    }
165
166    fn schema_entry_by_name(&self, name: &str) -> &'a SchemaEntry {
167        self.schema
168            .columns()
169            .values()
170            .find(|entry| entry.name() == name)
171            .unwrap_or_else(|| panic!("schema does not contain column named {name}"))
172    }
173
174    fn value_matches_schema_entry(schema_entry: &SchemaEntry, value: &RowCellValue) -> bool {
175        matches!(
176            (schema_entry.data_type(), value),
177            (_, RowCellValue::NULL)
178                | (
179                    crate::schemas::data_type::DataType::Int,
180                    RowCellValue::Int(_)
181                )
182                | (
183                    crate::schemas::data_type::DataType::String,
184                    RowCellValue::String(_)
185                )
186        )
187    }
188}

190#[cfg(test)]
191mod row_builder_tests {
192    use super::*;
193    use crate::schemas::data_type::DataType;
194
195    #[test]
196    fn test_row_builder_creates_cells_from_column_ids() {
197        let schema = Schema::builder(1)
198            .column(0, "id", DataType::Int, 0, true)
199            .column(1, "name", DataType::String, 1, false)
200            .pk([0])
201            .build()
202            .unwrap();
203
204        let row = Row::builder(&schema).value(0, 10).value(1, "Ada").build();
205
206        let mut values = row.values().collect::<Vec<_>>();
207        values.sort_by_key(|cell| schema.columns()[&cell.schema_entry_id()].position());
208
209        assert_eq!(values.len(), 2);
210        assert_eq!(values[0].schema_entry_id().owner_id(), 1);
211        assert_eq!(values[0].schema_entry_id().column_id(), 0);
212        assert!(matches!(values[0].value(), RowCellValue::Int(10)));
213        assert_eq!(values[1].schema_entry_id().owner_id(), 1);
214        assert_eq!(values[1].schema_entry_id().column_id(), 1);
215        assert!(matches!(values[1].value(), RowCellValue::String(value) if value == "Ada"));
216    }

218    #[test]
219    fn test_row_builder_creates_cells_from_schema_entries_positions_and_names() {
220        let schema = Schema::builder(1)
221            .column(0, "id", DataType::Int, 0, true)
222            .column(1, "name", DataType::String, 1, false)
223            .column(2, "email", DataType::String, 2, false)
224            .pk([0])
225            .build()
226            .unwrap();
227
228        let id_entry = schema
229            .columns()
230            .values()
231            .find(|entry| entry.name() == "id")
232            .unwrap();
233
234        let row = Row::builder(&schema)
235            .cell(id_entry, 10)
236            .value_named("name", "Ada")
237            .value_at_position(2, RowCellValue::NULL)
238            .build();
239
240        let mut values = row.values().collect::<Vec<_>>();
241        values.sort_by_key(|cell| schema.columns()[&cell.schema_entry_id()].position());
242
243        assert_eq!(values.len(), 3);
244        assert!(matches!(values[0].value(), RowCellValue::Int(10)));
245        assert!(matches!(values[1].value(), RowCellValue::String(value) if value == "Ada"));
246        assert!(matches!(values[2].value(), RowCellValue::NULL));
247    }
