4/// A wrapper type around the concept of the null bitmap, backed by a `u64`.
5#[derive(Debug)]
6pub struct Bitmap {
7    value: u64,
8}
9
10impl Default for Bitmap {
11    fn default() -> Self {
12        Self::new()
13    }
14}
15
16impl Bitmap {
17    // region CONSTRUCTORS
18    /// Create an empty bitmap
19    pub fn new() -> Self {
20        Self { value: 0 }
21    }
22
23    /// Creates a new `Bitmap` from a provided `u64`
24    pub fn from_u64(value: u64) -> Self {
25        Self { value }
26    }
27
28    /// Creates a new `Bitmap` from a provided byte slice
29    pub fn from_bytes(bytes: &[u8]) -> Result<Self, BinaryError> {
30        let converted = u64::from_le_bytes(bytes.try_into()?);
31        Ok(Self::from_u64(converted))
32    }
33    // endregion
34
35    // region GETTERS
36
37    /// Returns the bitmap as a byte array
38    pub fn to_bytes(&self) -> Result<Vec<u8>, BinaryError> {
39        let converted = u64::to_le_bytes(self.value);
40        Ok(converted.to_vec())
41    }
42
43    // endregion
44
45    /// Toggles the flag found at the provided index. Modification done in-place
46    /// ## Parameters
47    /// - `index`: the 0 indexed position of the flag to be flipped.
48    pub fn toggle(&mut self, index: usize) {
49        let mask = 1 << index;
50        self.value ^= mask;
51    }
52
53    /// Sets the flag from the provided position to "enabled". Modification done in place.
54    /// ## Parameters
55    /// - `index`: the 0 indexed position of the flag to be enabled.
56    pub fn set(&mut self, index: usize) {
57        let mask = 1 << index;
58        self.value |= mask;
59    }
60
61    /// Sets the flag from the provided position to "disabled". Modification done in place.
62    /// ## Parameters
63    /// - `index`: the 0 indexed position of the flag to be disabled.
64    pub fn unset(&mut self, index: usize) {
65        let mask = 1 << index;
66        let negated_mask = !mask;
67        self.value &= negated_mask;
68    }
69
70    /// Checks whether the value of the flag at the provided position is enabled or not.
71    /// ## Parameters
72    /// - `index`: the 0 indexed position of the flag to be disabled.
73    ///
74    /// ## Returns
75    /// - `true` if the flag is set, `false` if not set
76    pub fn check(&self, index: usize) -> bool {
77        let mask = 1 << index;
78        (self.value & mask) == mask
79    }
80}

82#[cfg(test)]
83mod test {

86    #[test]
87    pub fn toggle_first_correctly_flips_flag() {
88        let mut bitmap = Bitmap::new();
89        bitmap.toggle(0);
90        assert_eq!(bitmap.value, 1);
91        bitmap.toggle(0);
92        assert_eq!(bitmap.value, 0);
93    }
94
95    #[test]
96    pub fn toggle_leaves_others_unaffected() {
97        // 5 in binary is 0101
98        let mut bitmap = Bitmap::from_u64(5u64);
99        bitmap.toggle(1);
100        // we expect to find 0111 (7)
101        assert_eq!(bitmap.value, 7u64);
102    }
103
104    #[test]
105    pub fn set_correctly_enables_flag_and_leaves_others_unaffected() {
106        let mut bitmap = Bitmap::from_u64(5u64);
107        // 5 in binary is 0101
108        bitmap.set(1);
109        // we expect to find 0111 (7)
110        assert_eq!(bitmap.value, 7u64);
111    }
112
113    #[test]
114    pub fn unset_correctly_disables_flag_and_leaves_others_unaffected() {
115        let mut bitmap = Bitmap::from_u64(7u64);
116        // 7 in binary is 0111
117        bitmap.unset(1);
118        // we expect to find 0101 (5)
119        assert_eq!(bitmap.value, 5u64);
120    }
121
122    #[test]
123    pub fn check_correctly_returns_flag_value() {
124        let bitmap = Bitmap::from_u64(5u64);

126        assert!(bitmap.check(0));
